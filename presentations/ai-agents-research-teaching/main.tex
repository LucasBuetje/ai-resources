\section{What is an AI agent?}
% ══════════════════════════════════════════════════════

\begin{frame}[fragile]{You open an agent inside a folder: that folder is its world}

\vfill
You launch the agent \emph{in} a project folder. Everything happens there.

\vspace{0.6em}
\begin{columns}[T]
  \begin{column}{0.56\textwidth}
\begin{lstlisting}
my-project/        <- you start the agent here
  CLAUDE.md          project context (the rules)
  log/current.md     memory across sessions
  data/  code/  ...  your files
\end{lstlisting}
  \end{column}
  \begin{column}{0.42\textwidth}
    \footnotesize
    \hint{\yes}{Reads, writes, acts \mk{inside this folder}}
    \hint{\warn}{Cannot reach files \mk{above} it}
  \end{column}
\end{columns}

\vfill
\centering
\textcolor{seeblau}{A collaborator that lives \emph{inside} one project, not a chatbot in the cloud.}

\end{frame}

\begin{frame}{We all use LLM chatbots. Agents can do more}

\vfill
\begin{columns}[T]
  \begin{column}{0.46\textwidth}
    \textbf{\textcolor{grau}{Chatbots}}\\[0.6em]
    A chat window. \\
    You type, the AI replies with text.\\[0.8em]
    {\scriptsize Claude\,\textbullet\,ChatGPT\,\textbullet\,Gemini\,\textbullet\,Mistral}\\[0.9em]
    \textcolor{grau}{\faTimes}\enspace No access to your computer\\[0.5em]
    \textcolor{grau}{\faTimes}\enspace No files, no commands\\[0.5em]
    \textcolor{grau}{\faTimes}\enspace Text in, text out
  \end{column}
  \hfill
  \begin{column}{0.46\textwidth}
    \textbf{\textcolor{seeblau}{Agentic AI}}\\[0.6em]
    A program on your computer.\\
    The AI acts on its own.\\[0.8em]
    {\scriptsize Claude Code\,\textbullet\,Codex\,\textbullet\,Antigravity\,\textbullet\,OpenCode}\\[0.9em]
    \yes\enspace Reads and writes files\\[0.5em]
    \yes\enspace Runs terminal commands\\[0.5em]
    \yes\enspace Completes multi-step tasks
  \end{column}
\end{columns}
\vfill

\end{frame}


\begin{frame}{An agent can do anything your computer can}

\vfill
\begin{itemize}\setlength{\itemsep}{0.9em}
  \item \textbf{Read and write files:} text, code, PDF, spreadsheets
  \item \textbf{Run code:} Python, R, Bash, compile \LaTeX
  \item \textbf{Terminal commands:} install packages, Git, any system command
  \item \textbf{Search the web:} read documentation, check sources
  \item \textbf{Multi-step tasks:} plan, execute, fix errors, keep going
\end{itemize}
\vfill
\hint{\warn}{In principle, the AI can do anything on your computer that an IT expert could.}
\vfill

\end{frame}





\begin{frame}{That power brings real risks}

\begin{columns}[T]
  \begin{column}{0.48\textwidth}
    \textbf{Risks}\\[0.5em]
    \small
    \hint{\warn}{File access is \textbf{real}: files can be overwritten or deleted}
    \hint{\warn}{Plugins (MCPs) from unknown sources can inject malicious code}
    \hint{\warn}{Sensitive data in the working folder is readable by the AI}
  \end{column}
  \begin{column}{0.48\textwidth}
    \textbf{Workflow implications}\\[0.5em]
    \small
    \hint{\yes}{\textbf{Plan mode} first: review changes before they run}
    \hint{\yes}{\textbf{Back up} important files regularly}
    \hint{\yes}{\textbf{Install} only what you \textbf{trust}}
    \hint{\yes}{\textbf{No sensitive data} in the working folder}

  \end{column}
\end{columns}

\end{frame}


\begin{frame}{The tools guard you, but never completely}

\small
Agents like \textbf{Claude Code}, \textbf{Codex}, \textbf{Antigravity} and \textbf{OpenCode} build in safeguards:\\[0.6em]
\hint{\yes}{\textbf{Permission before each action:} before a file is written or a command runs, it asks; you see exactly what is about to happen}
\hint{\yes}{\textbf{Plan / read-only mode:} the AI thinks first, you approve}
\hint{\yes}{\textbf{Confined to the working folder:} no access above the opened project folder}
\hint{\yes}{\textbf{Changes are visible:} file diffs are shown before they are applied}
\vfill
\textbf{Stay alert anyway:}\\[0.3em]
\hint{\warn}{LLMs use \textbf{randomness}: same prompt, different output; sometimes a wrong one}
\hint{\warn}{The tools are \textbf{not perfect} and keep changing. Safeguards can fail.}
\hint{\warn}{Permission pop-ups are quickly \textbf{clicked away}. Do not confirm blindly.}

\end{frame}

\section{How to use an agent well \\ (in my experience)}
% ══════════════════════════════════════════════════════

\begin{frame}[standout]
  \vfill
  {\large None of the following requires manual setup on your part.}

  \vspace{1.2em}
  \normalsize
  Just ask the agent:\\[0.6em]
  \enquote{Set up a \texttt{CLAUDE.md}} \\
  \enquote{Create a skill based on this github link, adapted for my workflow.} \\
  \enquote{Use one subagent per paper.}
  \vfill
\end{frame}


\begin{frame}{An agent is a brilliant RA --- with no context and no memory}

\centering
\large
Think of it as a \mk{highly capable, fast, tireless research assistant}.\\[1.0em]
\normalsize

\begin{columns}[T]
  \begin{column}{0.48\textwidth}
    \textbf{Without context}\\[0.4em]
    \small
    \hint{\warn}{It doesn't know your project, your conventions, your data}
    \hint{\warn}{So it guesses (\,$\rightarrow$ errors\,) or works it out the slow way (\,$\rightarrow$ wasted time\,)}
  \end{column}
  \begin{column}{0.48\textwidth}
    \textbf{Without memory}\\[0.4em]
    \small
    \hint{\warn}{It forgets everything between sessions}
    \hint{\warn}{Every new session is a \textbf{fresh} RA, starting again from day one}
  \end{column}
\end{columns}

\vfill
\centering
{\large\textcolor{seeblau}{So give it what any good RA needs: \mk{context} and \mk{memory}.}}
\vfill

\end{frame}


\begin{frame}[fragile]{1. \texttt{CLAUDE.md} gives the agent your context, once}

\small
A plain Markdown file the agent reads at the \emph{start of every session}. Two levels, and they stack:

\vspace{0.6em}
\begin{columns}[T]
  \begin{column}{0.49\textwidth}
    \textbf{\textcolor{seeblau}{Global}} \quad {\footnotesize\texttt{\textasciitilde/.claude/CLAUDE.md}}\\[0.4em]
    Applies \emph{everywhere}: \mk{how you work}.\\[0.5em]
    \footnotesize
    \textbullet\ Never overwrite raw data\\
    \textbullet\ Don't make research design decisions without me\\
    \textbullet\ German writing conventions
  \end{column}
  \begin{column}{0.49\textwidth}
    \textbf{\textcolor{seeblau}{Local}} \quad {\footnotesize\texttt{<project>/CLAUDE.md}}\\[0.4em]
    Applies \emph{here} only: \mk{what this project is}.\\[0.5em]
    \footnotesize
    \textbullet\ Data lives in \texttt{data/wave6/}\\
    \textbullet\ This deck: build with the \texttt{deck} skill\\
    \textbullet\ Codebook in \texttt{docs/codebook.md}
  \end{column}
\end{columns}

\vfill
\hint{\faLightbulb}{\textbf{Why it matters:} write it once and the agent follows it forever. You stop re-explaining how you work, and how \emph{this} project is laid out, at the top of every chat.}

\end{frame}


\begin{frame}[fragile]{2. A running log gives the agent memory across sessions}

\small
Models forget everything between sessions. A habit I adopted from \textbf{Scott Cunningham}: keep one working-memory file, \texttt{log/current.md}, and let the agent maintain it.

\vspace{0.5em}
\textbf{The loop:}\\[0.1em]
\hint{\textcolor{seeblau}{\textbf{1}}}{\textbf{As agent works:} the agent appends short entries: what changed, why, what is still open}
\hint{\textcolor{seeblau}{\textbf{2}}}{\textbf{At the next start:} the file is read back in automatically, so it resumes where you stopped}
\hint{\textcolor{seeblau}{\textbf{3}}}{Say \mk{\enquote{log}}: archive \texttt{current.md} to a timestamped file, start a fresh one}

\vspace{0.3em}
\begin{lstlisting}
2026-06-24 -- Cleaned wave 6, merged on pidp. Open: 142
  duplicate household ids -- decide merge key next time.
\end{lstlisting}

\end{frame}


\begin{frame}{3. Plan mode lets you approve before anything happens}

\small
The agent investigates and writes a \textbf{plan}, but changes \emph{nothing} until you say go. The single most useful safety habit.

\vfill
\begin{columns}[T]
  \begin{column}{0.48\textwidth}
    \textbf{Without plan mode}\\[0.4em]
    \hint{\warn}{It starts editing immediately}
    \hint{\warn}{You discover a wrong assumption \emph{after} 12 files changed}
  \end{column}
  \begin{column}{0.48\textwidth}
    \textbf{With plan mode}\\[0.4em]
    \hint{\yes}{You read the plan, fix the misunderstanding}
    \hint{\yes}{Only then does it touch your files}
  \end{column}
\end{columns}

\vfill
\hint{\liveicon}{\textbf{Live demo:} Let's try creating a set of exercises to practice what this talk covers}

\end{frame}


\begin{frame}{4. Skills package expertise you reuse}

\small
A skill is a reusable, named recipe: expert instructions you (or others) wrote once, invoked with one command.

\vfill
\begin{columns}[T]
  \begin{column}{0.5\textwidth}
    \textbf{Research}\\[0.3em]
    \hint{\faBookOpen}{\texttt{/summarize-paper}: structured summary}
    \hint{\faCheckDouble}{\texttt{/referee2}: audit a complete code pipeline}
    \hint{\faGlasses}{\texttt{/blindspot}: find what you can't see in your own results}
  \end{column}
  \begin{column}{0.5\textwidth}
    \textbf{Teaching \& craft}\\[0.3em]
    \hint{\faChalkboardTeacher}{\texttt{/quiz-review}: check a quiz for correctness}
    \hint{\faLayerGroup}{\texttt{/deck}: build a deck (this one!)}
    \hint{\faLanguage}{\texttt{/germancheck}: catch translation tells}
  \end{column}
\end{columns}

\vfill
\hint{\faGraduationCap}{\textbf{Example:} \texttt{/summarize-paper} reads a paper end-to-end and returns a structured extract}

\end{frame}


\begin{frame}{5. Subagents turn one assistant into a team}

\small
The agent can spawn \textbf{many subagents} that work in parallel, each on a slice, then report back. 



\vfill
\begin{itemize}
  \item \textbf{Reading list:} \texttt{/summarize-paper} fans out to multiple papers in parallel; done in the time it takes to read one
  \item \textbf{Paper audit:} \texttt{/referee2} runs five specialists simultaneously: code, replication, econometrics, structure, prose
  \item \textbf{LLM Council:} \texttt{/council} sends a question to four models; they critique each other and a chair synthesises
\end{itemize}

\vfill
\hint{\faGraduationCap}{\textbf{Example:} 30 papers on peer effects summarized and synthesized in a coffee break}

\end{frame}

\begin{frame}{6. MCP connects the agent to other tools (like Zotero)}

\small (\textbf{M}odel \textbf{C}ontext \textbf{P}rotocol) is a standard plug for external tools. Each has to be installed once.

\vfill
\begin{columns}[T]
  \begin{column}{0.5\textwidth}
    \textbf{With the Zotero MCP, I can:}\\[0.4em]
    \hint{\faSearch}{Ask:\enquote{What papers do I have on X?}}
    \hint{\faFilePdf}{Let Claude get the PDF directly from Zotero (incl. my highlights/notes)}
  \end{column}
  \begin{column}{0.5\textwidth}
    \textbf{Caution}\\[0.4em]
    \hint{\warn}{An MCP runs code on your machine. Install only what you trust.}
  \end{column}
\end{columns}

\vfill
\hint{\liveicon}{\textbf{Live demo:} Let's try cleaning up some unstructured Zotero library}

\end{frame}


\begin{frame}[fragile]{7. Hooks make the agent enforce your conventions}

\small
A hook is a small script the tool runs automatically at set moments: \emph{before} a command, \emph{after} an edit, at session start. Your rules, enforced by code, not by trust.

\begin{lstlisting}
# Before any Bash command: block `rm`, suggest archiving
guard-rm.sh        -> "use: mv <file> archive/"

# At session start: read log/current.md and brief the agent
session-start.sh   -> summarise what was in progress
\end{lstlisting}

\vfill
\hint{\faGraduationCap}{\textbf{Example:} the agent opens with a one-line briefing on where the last session ended, without me typing anything.}

\end{frame}


\begin{frame}{8. Verification is the new bottleneck}

\vfill
\centering\small\textcolor{grau}{\enquote{Verification is the new bottleneck. Production used to be the bottleneck of research, now verification is.} --- Scott Cunningham}

\vfill
\begin{columns}[T]
  \begin{column}{0.48\textwidth}
    \textbf{The problem}\\[0.4em]
    \hint{\warn}{Code, analyses, even papers are now almost costless to produce}
    \hint{\warn}{The agent is confident even when it is wrong; it rarely says \enquote{I'm not sure}}
  \end{column}
  \begin{column}{0.48\textwidth}
    \textbf{Agents can help here too}\\[0.4em]
    \hint{\yes}{Write verification scripts, re-runnable, kept with the project}
    \hint{\yes}{Let the agent check its output against a known truth (e.g. with MC simulations)}
  \end{column}
\end{columns}

\vfill
\hint{\faGraduationCap}{\textbf{Example:} I had to slightly adapt a function from an R package. Claude wrote verification scripts to make sure the estimator still recovers the true value}

\end{frame}

\section{The reveal + a note}
% ══════════════════════════════════════════════════════



\begin{frame}[fragile]{Nothing is lost: every session is on your disk}

\small
The whole run is preserved as structured text. The work is \mk{reproducible and auditable}.

\begin{lstlisting}
~/.claude/projects/<working-dir>/<session-id>.jsonl
\end{lstlisting}

\vspace{-0.3em}
\small
Inside: your original prompt (typos and all), every tool call, every subagent, every R script, every error and retry. A \enquote{flight recorder} for the work.

\vfill
\hint{\faLightbulb}{\textbf{Why it matters:} the quiet fear is that AI work is a black box. It is not. The full record sits on your machine, ready to send to a co-author or point another agent at.}

\end{frame}
